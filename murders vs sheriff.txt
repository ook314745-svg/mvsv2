local ESP_Enabled = false
local Hitbox_Enabled = false
local Hitbox_Size = 10
local ESP_Tag = "Z_ESP"
local Connection = nil


local function CreateESP(plr)
    if plr == LocalPlayer then return end
    
    local function UpdateChar()
        local char = plr.Character
        if not char then return end
        local head = char:FindFirstChild("Head")
        if not head then return end
        
        if head:FindFirstChild(ESP_Tag) then return end
        
        local bill = Instance.new("BillboardGui")
        bill.Name = ESP_Tag
        bill.Adornee = head
        bill.Size = UDim2.new(0, 200, 0, 50)
        bill.StudsOffset = Vector3.new(0, 2, 0)
        bill.AlwaysOnTop = true
        
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.Text = plr.Name
        label.TextColor3 = Color3.fromRGB(255, 0, 85)
        label.Font = Enum.Font.GothamBold
        label.TextSize = 14
        label.TextStrokeTransparency = 0.5
        label.Parent = bill
        
        bill.Parent = head
    end
    
    UpdateChar()
    plr.CharacterAdded:Connect(function()
        wait(0.5)
        if ESP_Enabled then UpdateChar() end
    end)
end


local function UpdateHitbox(plr)
    if plr == LocalPlayer then return end
    
    local char = plr.Character
    if not char then return end
    
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if hrp then
        if Hitbox_Enabled then
            hrp.Size = Vector3.new(Hitbox_Size, Hitbox_Size, Hitbox_Size)
            hrp.Transparency = 0.7
            hrp.Color = Color3.fromRGB(255, 0, 85)
            hrp.CanCollide = false
        else
            hrp.Size = Vector3.new(2, 2, 1)
            hrp.Transparency = 1
        end
    end
end


ESPBtn.MouseButton1Click:Connect(function()
    ESP_Enabled = not ESP_Enabled
    
    if ESP_Enabled then
        ESPBtn.Text = "ESP: ON"
        ESPBtn.BackgroundColor3 = Color3.fromRGB(255, 0, 85)
        
        for _, plr in pairs(Players:GetPlayers()) do
            CreateESP(plr)
        end
        
        
        Connection = RunService.Heartbeat:Connect(function()
            for _, plr in pairs(Players:GetPlayers()) do
                if plr.Character and plr.Character:FindFirstChild("Head") then
                    if not plr.Character.Head:FindFirstChild(ESP_Tag) then
                        CreateESP(plr)
                    end
                end
            end
        end)
    else
        ESPBtn.Text = "ESP: OFF"
        ESPBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
        
        if Connection then
            Connection:Disconnect()
            Connection = nil
        end
        
        
        for _, plr in pairs(Players:GetPlayers()) do
            if plr.Character and plr.Character:FindFirstChild("Head") then
                local tag = plr.Character.Head:FindFirstChild(ESP_Tag)
                if tag then tag:Destroy() end
            end
        end
    end
end)


HitboxBtn.MouseButton1Click:Connect(function()
    Hitbox_Enabled = not Hitbox_Enabled
    
    if Hitbox_Enabled then
        HitboxBtn.Text = "HITBOX: ON"
        HitboxBtn.BackgroundColor3 = Color3.fromRGB(255, 0, 85)
    else
        HitboxBtn.Text = "HITBOX: OFF"
        HitboxBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    end
    
    for _, plr in pairs(Players:GetPlayers()) do
        UpdateHitbox(plr)
    end
end)


Players.PlayerAdded:Connect(function(plr)
    plr.CharacterAdded:Connect(function()
        wait(0.5)
        if Hitbox_Enabled then UpdateHitbox(plr) end
        if ESP_Enabled then CreateESP(plr) end
    end)
end)


local dragging = false
local sliderWidth = SliderBar.AbsoluteSize.X

local function updateSlider(input)
    local pos = math.clamp((input.Position.X - SliderBar.AbsolutePosition.X) / sliderWidth, 0, 1)
    SliderFill.Size = UDim2.new(pos, 0, 1, 0)
    SliderKnob.Position = UDim2.new(pos, -8, 0.5, -8)
    
    Hitbox_Size = math.floor(5 + (pos * 45))
    SizeLabel.Text = "HITBOX SIZE: " .. Hitbox_Size
    
    if Hitbox_Enabled then
        for _, plr in pairs(Players:GetPlayers()) do
            UpdateHitbox(plr)
        end
    end
end

SliderKnob.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        updateSlider(input)
    end
end)

SliderBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        updateSlider(input)
    end
end)


RunService.Heartbeat:Connect(function()
    if Hitbox_Enabled then
        for _, plr in pairs(Players:GetPlayers()) do
            if plr.Character then
                local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
                if hrp and hrp.Size.X ~= Hitbox_Size then
                    UpdateHitbox(plr)
                end
            end
        end
    end
end)
